```python
import numpy as np

# ===== 激活函数 =====
def sigmoid(Z):
    return 1 / (1 + np.exp(-Z))

def sigmoid_derivative(A):
    return A * (1 - A)

def relu(Z):
    return np.maximum(0, Z)

def relu_derivative(A):
    return (A > 0).astype(float)

def softmax(Z):
    exp_Z = np.exp(Z - np.max(Z, axis=0, keepdims=True))  # 稳定性
    return exp_Z / np.sum(exp_Z, axis=0, keepdims=True)


# ===== Layer 层 =====
class Layer:
    def __init__(self, input_size, output_size, activation="relu"):
        self.W = np.random.randn(output_size, input_size) * 0.01
        self.b = np.zeros((output_size, 1))
        self.activation = activation

    def forward(self, A_prev):
        """ 前向传播 """
        self.A_prev = A_prev
        self.Z = np.dot(self.W, A_prev) + self.b
        
        if self.activation == "sigmoid":
            self.A = sigmoid(self.Z)
        elif self.activation == "relu":
            self.A = relu(self.Z)
        elif self.activation == "softmax":
            self.A = softmax(self.Z)
        else:
            raise ValueError("Unsupported activation")
        return self.A

    def backward(self, dA, is_output=False, Y=None):
        """ 反向传播 """
        m = self.A_prev.shape[1]

        if self.activation == "softmax" and is_output:
            # 输出层 softmax+交叉熵梯度简化
            dZ = self.A - Y
        else:
            if self.activation == "sigmoid":
                dZ = dA * sigmoid_derivative(self.A)
            elif self.activation == "relu":
                dZ = dA * relu_derivative(self.A)
            else:
                raise ValueError("Unsupported activation")

        # 计算梯度
        self.dW = (1/m) * np.dot(dZ, self.A_prev.T)
        self.db = (1/m) * np.sum(dZ, axis=1, keepdims=True)
        dA_prev = np.dot(self.W.T, dZ)
        return dA_prev

    def update_params(self, lr):
        self.W -= lr * self.dW
        self.b -= lr * self.db


# ===== Neural Network =====
class NeuralNetwork:
    def __init__(self, layers_config):
        """
        layers_config: list of (input_size, output_size, activation)
        """
        self.layers = []
        for (in_size, out_size, act) in layers_config:
            self.layers.append(Layer(in_size, out_size, act))

    def forward(self, X):
        A = X
        for layer in self.layers:
            A = layer.forward(A)
        return A

    def compute_loss(self, Y_hat, Y):
        m = Y.shape[1]
        loss = -np.sum(Y * np.log(Y_hat + 1e-9)) / m
        return loss

    def backward(self, Y):
        dA = None
        for i in reversed(range(len(self.layers))):
            layer = self.layers[i]
            if i == len(self.layers) - 1:
                dA = layer.backward(dA, is_output=True, Y=Y)
            else:
                dA = layer.backward(dA)

    def update(self, lr):
        for layer in self.layers:
            layer.update_params(lr)

    def train(self, X, Y, epochs=1000, lr=0.01):
        for epoch in range(epochs):
            Y_hat = self.forward(X)
            loss = self.compute_loss(Y_hat, Y)
            self.backward(Y)
            self.update(lr)

            if epoch % 100 == 0:
                print(f"Epoch {epoch}, Loss: {loss:.4f}")

    def predict(self, X):
        Y_hat = self.forward(X)
        return np.argmax(Y_hat, axis=0)

    # 假设输入维度=4, 分类数=3
    X = np.random.randn(4, 200)  # (features, samples)
    Y = np.random.randint(0, 3, 200)
    Y_onehot = np.eye(3)[Y].T   # (3, 200)

    # 构建网络: [4->10->8->3]
    layers_config = [
        (4, 10, "relu"),
        (10, 8, "sigmoid"),
        (8, 3, "softmax")
    ]

    nn = NeuralNetwork(layers_config)
    nn.train(X, Y_onehot, epochs=1000, lr=0.1)

    preds = nn.predict(X)
    acc = np.mean(preds == Y)
    print(f"训练集准确率: {acc:.2f}")
```

```python
# -------------------- 激活函数 --------------------
def sigmoid(Z):
    return 1 / (1 + np.exp(-Z))

def sigmoid_grad(A):
    return A * (1 - A)

def relu(Z):
    return np.maximum(0, Z)

def relu_grad(A):
    return (A > 0).astype(float)

def softmax(Z):
    expZ = np.exp(Z - np.max(Z, axis=0, keepdims=True))  # 防止溢出
    return expZ / np.sum(expZ, axis=0, keepdims=True)


# -------------------- Layer 层 --------------------
class Layer:
    def __init__(self, in_dim, out_dim, activation="relu"):
        self.in_dim = in_dim
        self.out_dim = out_dim
        self.activation = activation
        
        # Kaiming 初始化 (He)
        if activation == "relu":
            self.W = np.random.randn(out_dim, in_dim) * np.sqrt(2.0 / in_dim)
        else:  # sigmoid 用 Xavier 更好，但这里统一用 He
            self.W = np.random.randn(out_dim, in_dim) * np.sqrt(2.0 / in_dim)
        self.b = np.zeros((out_dim, 1))

    def forward(self, A_prev):
        self.A_prev = A_prev  # (in_dim, m)
        self.Z = np.dot(self.W, A_prev) + self.b
        if self.activation == "sigmoid":
            self.A = sigmoid(self.Z)
        elif self.activation == "relu":
            self.A = relu(self.Z)
        else:
            raise ValueError("Unsupported activation")
        return self.A

    def backward(self, dA):
        m = self.A_prev.shape[1]
        # 激活函数导数
        if self.activation == "sigmoid":
            dZ = dA * sigmoid_grad(self.A)
        elif self.activation == "relu":
            dZ = dA * relu_grad(self.A)
        else:
            raise ValueError("Unsupported activation")
        # 反向传播
        self.dW = (1 / m) * np.dot(dZ, self.A_prev.T)
        self.db = (1 / m) * np.sum(dZ, axis=1, keepdims=True)
        dA_prev = np.dot(self.W.T, dZ)
        return dA_prev

    def update(self, lr):
        self.W -= lr * self.dW
        self.b -= lr * self.db

class NeuralNetwork:
    def __init__(self, layers_dims, activations):
        assert len(layers_dims) - 1 == len(activations)
        self.layers = []
        for i in range(len(activations)):
            self.layers.append(Layer(layers_dims[i], layers_dims[i+1], activations[i]))

    def forward(self, X):
        A = X
        for layer in self.layers[:-1]:
            A = layer.forward(A)
        # 最后一层 softmax
        ZL = np.dot(self.layers[-1].W, A) + self.layers[-1].b
        self.layers[-1].Z = ZL
        self.layers[-1].A_prev = A
        AL = softmax(ZL)
        self.layers[-1].A = AL
        return AL

    def compute_loss(self, Y_hat, Y):
        m = Y.shape[1]
        loss = -np.sum(Y * np.log(Y_hat + 1e-8)) / m
        return loss

    def backward(self, Y_hat, Y):
        m = Y.shape[1]
        dZL = Y_hat - Y  # softmax+交叉熵
        last_layer = self.layers[-1]
        last_layer.dW = (1 / m) * np.dot(dZL, last_layer.A_prev.T)
        last_layer.db = (1 / m) * np.sum(dZL, axis=1, keepdims=True)
        dA_prev = np.dot(last_layer.W.T, dZL)

        for layer in reversed(self.layers[:-1]):
            dA_prev = layer.backward(dA_prev)

    def update(self, lr):
        for layer in self.layers:
            layer.update(lr)

    def fit(self, X, Y, epochs=100, lr=0.01, batch_size=32):
        m = X.shape[1]
        for epoch in range(epochs):
            perm = np.random.permutation(m)
            X_shuffled = X[:, perm]
            Y_shuffled = Y[:, perm]

            for i in range(0, m, batch_size):
                X_batch = X_shuffled[:, i:i+batch_size]
                Y_batch = Y_shuffled[:, i:i+batch_size]

                Y_hat = self.forward(X_batch)
                loss = self.compute_loss(Y_hat, Y_batch)
                self.backward(Y_hat, Y_batch)
                self.update(lr)

            if (epoch+1) % 10 == 0:
                Y_hat_full = self.forward(X)
                loss_full = self.compute_loss(Y_hat_full, Y)

    def predict(self, X):
```

```python
import numpy as np

# ========== 激活函数 ==========
def sigmoid(z):
    # 防止溢出：clip z 到 [-500, 500]
    z = np.clip(z, -500, 500)
    return 1 / (1 + np.exp(-z))

def sigmoid_derivative(a):
    return a * (1 - a)

def relu(z):
    return np.maximum(0, z)

def relu_derivative(a):
    return (a > 0).astype(float)

def softmax(z):
    z = z - np.max(z, axis=0, keepdims=True)  # 防止exp溢出
    exp_scores = np.exp(z)
    return exp_scores / np.sum(exp_scores, axis=0, keepdims=True)


# ========== Layer 层 ==========
class Layer:
    def __init__(self, input_size, output_size, activation="relu"):
        # Kaiming 初始化 (He initialization)
        self.W = np.random.randn(output_size, input_size) * np.sqrt(2.0 / input_size)
        self.b = np.zeros((output_size, 1))
        self.activation = activation

    def forward(self, A_prev):
        self.Z = np.dot(self.W, A_prev) + self.b
        if self.activation == "sigmoid":
            self.A = sigmoid(self.Z)
        elif self.activation == "relu":
            self.A = relu(self.Z)
        elif self.activation == "softmax":
            self.A = softmax(self.Z)
        return self.A

    def backward(self, dA, A_prev):
        m = A_prev.shape[1]

        if self.activation == "sigmoid":
            dZ = dA * sigmoid_derivative(self.A)
        elif self.activation == "relu":
            dZ = dA * relu_derivative(self.A)
        else:
            dZ = dA

        self.dW = np.dot(dZ, A_prev.T) / m
        self.db = np.sum(dZ, axis=1, keepdims=True) / m
        dA_prev = np.dot(self.W.T, dZ)

        return dA_prev


# ========== Neural Network with Adam ==========
class NeuralNetwork:
    def __init__(self, layers):
        self.layers = layers
        # Adam 参数
        self.vdW = [np.zeros_like(layer.W) for layer in self.layers]
        self.vdb = [np.zeros_like(layer.b) for layer in self.layers]
        self.sdW = [np.zeros_like(layer.W) for layer in self.layers]
        self.sdb = [np.zeros_like(layer.b) for layer in self.layers]
        self.t = 0  # 时间步数

    def forward(self, X):
        A = X
        for layer in self.layers:
            A = layer.forward(A)
        return A

    def compute_loss(self, Y_hat, Y):
        m = Y.shape[1]
        loss = -np.sum(Y * np.log(Y_hat + 1e-9)) / m
        return loss

    def backward(self, X, Y):
        m = X.shape[1]
        L = len(self.layers)

        dA = self.layers[-1].A - Y
        for l in reversed(range(L)):
            A_prev = X if l == 0 else self.layers[l-1].A
            dA = self.layers[l].backward(dA, A_prev)

    def update_parameters_adam(self, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.t += 1
        for l, layer in enumerate(self.layers):
            # 一阶矩估计
            self.vdW[l] = beta1 * self.vdW[l] + (1 - beta1) * layer.dW
            self.vdb[l] = beta1 * self.vdb[l] + (1 - beta1) * layer.db
            # 二阶矩估计
            self.sdW[l] = beta2 * self.sdW[l] + (1 - beta2) * (layer.dW ** 2)
            self.sdb[l] = beta2 * self.sdb[l] + (1 - beta2) * (layer.db ** 2)
            # 偏差修正
            vdW_corr = self.vdW[l] / (1 - beta1 ** self.t)
            vdb_corr = self.vdb[l] / (1 - beta1 ** self.t)
            sdW_corr = self.sdW[l] / (1 - beta2 ** self.t)
            sdb_corr = self.sdb[l] / (1 - beta2 ** self.t)
            # 更新参数
            layer.W -= lr * vdW_corr / (np.sqrt(sdW_corr) + eps)
            layer.b -= lr * vdb_corr / (np.sqrt(sdb_corr) + eps)

    def train(self, X, Y, epochs=1000, batch_size=64, lr=0.001):
        m = X.shape[1]
        for epoch in range(epochs):
            permutation = np.random.permutation(m)
            X_shuffled = X[:, permutation]
            Y_shuffled = Y[:, permutation]

            for i in range(0, m, batch_size):
                X_batch = X_shuffled[:, i:i+batch_size]
                Y_batch = Y_shuffled[:, i:i+batch_size]

                Y_hat = self.forward(X_batch)
                self.backward(X_batch, Y_batch)
                self.update_parameters_adam(lr)

            if (epoch+1) % 100 == 0:
                loss = self.compute_loss(self.forward(X), Y)
                print(f"Epoch {epoch+1}, Loss: {loss:.4f}")
```
